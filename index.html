import sys
input = sys.stdin.readline

def solve():
    n, m, k = map(int, input().split())
    grid = []
    for i in range(n):
        grid.append(input().strip())

    # Внешние грани колонии
    outside = 2 * n * m + 2 * k * (n + m)

    # Грани между соседями внутри одного слоя (не сросшиеся)
    within = 0
    for i in range(n):
        for j in range(m - 1):
            c1, c2 = grid[i][j], grid[i][j + 1]
            if not (c1 == c2 and c1 in ('R', 'G')):
                within += 1
    for i in range(n - 1):
        for j in range(m):
            c1, c2 = grid[i][j], grid[i + 1][j]
            if not (c1 == c2 and c1 in ('R', 'G')):
                within += 1

    within_total = within * k

    # Грани между слоями: срастаются только синие
    count_B = sum(row.count('B') for row in grid)
    between = (k - 1) * (n * m - count_B)

    print(outside + within_total + between)

solve()
